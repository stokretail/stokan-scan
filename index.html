// ============================================================ // PERBAIKAN SCAN DOBEL // Tiap scan membawa ID unik (requestId). ID yang sama tidak diproses dua kali. // File ini DITAMBAHKAN ke proyek Apps Script, di samping Code.gs. // Di Code.gs, ganti nama "function doPost(e)" menjadi "function doPostLama(e)". // ============================================================ // Kolom E di Log Keluar dipakai untuk catatan manual (mis. nama toko), jadi // ID Scan dan Petugas ditaruh di kolom G dan H. var KOLOM_ID_LOG = 7; // kolom G = ID Scan var KOLOM_PETUGAS_LOG = 8; // kolom H = nama petugas // KUNCI RAHASIA: ganti tulisan di antara tanda kutip dengan kunci buatan Anda sendiri // (huruf dan angka saja, minimal 12 karakter, tanpa spasi). Kunci yang sama diisi di // pengaturan aplikasi scan di tiap HP. Jangan dibagikan dan jangan ditulis di GitHub. var KUNCI_RAHASIA = 'GANTI-DENGAN-KUNCI-ANDA'; function doPost(e) { var body; try { body = JSON.parse(e.postData.contents); } catch (err) { return jsonResponse({ success: false, message: 'Data tidak terbaca.' }); } // Cek kunci rahasia untuk semua permintaan if (KUNCI_RAHASIA === 'GANTI-DENGAN-KUNCI-ANDA') { return jsonResponse({ success: false, kunciSalah: true, message: 'Kunci rahasia belum diatur di script.' }); } if (String(body.kunci || '') !== KUNCI_RAHASIA) { return jsonResponse({ success: false, kunciSalah: true, message: 'Kunci rahasia salah. Cek di pengaturan aplikasi.' }); } // Lihat daftar warna/size: tidak mengubah stok if (body.lookupModel) { return jsonResponse(lookupModel(body.lookupModel)); } var requestId = String(body.requestId || '').trim(); var petugas = String(body.petugas || '').trim().slice(0, 30); var lock = LockService.getScriptLock(); try { lock.waitLock(20000); } catch (err) { return jsonResponse({ success: false, message: 'Server sedang sibuk, coba lagi.' }); } try { if (body.undoBarcode) { return jsonResponse(prosesKeluar(String(body.undoBarcode).trim(), -Number(body.qty || 1), requestId, petugas)); } return jsonResponse(prosesKeluar(String(body.barcode || '').trim(), Number(body.qty || 1), requestId, petugas)); } catch (err) { return jsonResponse({ success: false, message: 'Terjadi error: ' + err.message }); } finally { lock.releaseLock(); } } // delta positif = barang keluar, delta negatif = pembatalan function prosesKeluar(kode, delta, requestId, petugas) { if (!delta || isNaN(delta)) { return { success: false, message: 'Jumlah tidak valid.' }; } var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(NAMA_SHEET_STOK); if (!sheet) { return { success: false, message: 'Sheet stok tidak ditemukan. Cek NAMA_SHEET_STOK.' }; } var data = sheet.getDataRange().getValues(); var header = data[0]; var kodeCol = header.indexOf(KOLOM_KODE); var namaCol = header.indexOf(KOLOM_NAMA); var keluarCol = header.indexOf(KOLOM_KELUAR); var stokAkhirCol = header.indexOf(KOLOM_STOK_AKHIR); if (kodeCol === -1 || keluarCol === -1) { return { success: false, message: 'Nama kolom tidak cocok. Cek KOLOM_KODE / KOLOM_KELUAR.' }; } var baris = -1; for (var i = 1; i < data.length; i++) { if (String(data[i][kodeCol]).trim() === kode) { baris = i; break; } } if (baris === -1) { return { success: false, message: 'Kode barang "' + kode + '" tidak ditemukan di data.' }; } var nama = data[baris][namaCol]; var keluarSekarang = Number(data[baris][keluarCol]) || 0; var rowIndex = baris + 1; var logSheet = ambilLogSheet(ss); // Sudah pernah diproses? Kembalikan hasil tanpa mengubah stok lagi. if (requestId && sudahDiproses(logSheet, requestId)) { return { success: true, duplikat: true, nama: nama, keluar: keluarSekarang, stokAkhir: stokAkhirCol !== -1 ? sheet.getRange(rowIndex, stokAkhirCol + 1).getValue() : '' }; } var keluarBaru = delta < 0 ? Math.max(0, keluarSekarang + delta) : keluarSekarang + delta; // 1) catat log dulu (lengkap dengan ID), 2) baru ubah stok var barisLog = logSheet.getLastRow() + 1; logSheet.getRange(barisLog, 1, 1, 4).setValues([[new Date(), kode, (delta < 0 ? '[DIBATALKAN] ' : '') + nama, delta]]); logSheet.getRange(barisLog, KOLOM_ID_LOG, 1, 2).setValues([[requestId, petugas || '']]); try { sheet.getRange(rowIndex, keluarCol + 1).setValue(keluarBaru); SpreadsheetApp.flush(); } catch (err) { logSheet.deleteRow(barisLog); // stok gagal diubah, batalkan catatan log throw err; } return { success: true, nama: nama, keluar: keluarBaru, stokAkhir: stokAkhirCol !== -1 ? sheet.getRange(rowIndex, stokAkhirCol + 1).getValue() : '' }; } function ambilLogSheet(ss) { var logSheet = ss.getSheetByName(NAMA_SHEET_LOG); if (!logSheet) { logSheet = ss.insertSheet(NAMA_SHEET_LOG); logSheet.appendRow(['Waktu', 'Kode Barang', 'Nama', 'Jumlah']); logSheet.getRange(1, KOLOM_ID_LOG, 1, 2).setValues([['ID Scan', 'Petugas']]); } else { if (!logSheet.getRange(1, KOLOM_ID_LOG).getValue()) logSheet.getRange(1, KOLOM_ID_LOG).setValue('ID Scan'); if (!logSheet.getRange(1, KOLOM_PETUGAS_LOG).getValue()) logSheet.getRange(1, KOLOM_PETUGAS_LOG).setValue('Petugas'); } return logSheet; } // Cek ID di 1000 baris log terakhir (cukup untuk kiriman ulang karena jaringan) function sudahDiproses(logSheet, requestId) { var terakhir = logSheet.getLastRow(); if (terakhir < 2) return false; var mulai = Math.max(2, terakhir - 999); var ketemu = logSheet .getRange(mulai, KOLOM_ID_LOG, terakhir - mulai + 1, 1) .createTextFinder(requestId) .matchEntireCell(true) .findNext(); return ketemu !== null; }
